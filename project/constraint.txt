# System clock (onboard 100 MHz)
set_property PACKAGE_PIN N11 [get_ports clk]
set_property IOSTANDARD LVCMOS33 [get_ports clk]
create_clock -period 10.000 -name sys_clk -waveform {0 5} [get_ports clk]

# Reset (mapped to switch L5)
set_property PACKAGE_PIN L5 [get_ports rst]
set_property IOSTANDARD LVCMOS33 [get_ports rst]

# Push Buttons
set_property PACKAGE_PIN K13 [get_ports btn_dmk]
set_property IOSTANDARD LVCMOS33 [get_ports btn_dmk]
set_property PULLDOWN TRUE [get_ports btn_dmk]

set_property PACKAGE_PIN L14 [get_ports btn_admk]
set_property IOSTANDARD LVCMOS33 [get_ports btn_admk]
set_property PULLDOWN TRUE [get_ports btn_admk]

set_property PACKAGE_PIN M12 [get_ports btn_tvk]
set_property IOSTANDARD LVCMOS33 [get_ports btn_tvk]
set_property PULLDOWN TRUE [get_ports btn_tvk]

set_property PACKAGE_PIN L13 [get_ports btn_ntk]
set_property IOSTANDARD LVCMOS33 [get_ports btn_ntk]
set_property PULLDOWN TRUE [get_ports btn_ntk]

set_property PACKAGE_PIN M14 [get_ports btn_other]
set_property IOSTANDARD LVCMOS33 [get_ports btn_other]
set_property PULLDOWN TRUE [get_ports btn_other]

# VGA sync
set_property PACKAGE_PIN F14 [get_ports hsync]
set_property IOSTANDARD LVCMOS33 [get_ports hsync]

set_property PACKAGE_PIN H16 [get_ports vsync]
set_property IOSTANDARD LVCMOS33 [get_ports vsync]

# VGA RGB mapping
# rgb[11:8] -> R[3:0]
set_property PACKAGE_PIN E16 [get_ports {rgb[11]}]
set_property PACKAGE_PIN F13 [get_ports {rgb[10]}]
set_property PACKAGE_PIN F12 [get_ports {rgb[9]}]
set_property PACKAGE_PIN D15 [get_ports {rgb[8]}]
set_property IOSTANDARD LVCMOS33 [get_ports {rgb[11] rgb[10] rgb[9] rgb[8]}]

# rgb[7:4] -> G[3:0]
set_property PACKAGE_PIN H11 [get_ports {rgb[7]}]
set_property PACKAGE_PIN E15 [get_ports {rgb[6]}]
set_property PACKAGE_PIN F15 [get_ports {rgb[5]}]
set_property PACKAGE_PIN D16 [get_ports {rgb[4]}]
set_property IOSTANDARD LVCMOS33 [get_ports {rgb[7] rgb[6] rgb[5] rgb[4]}]

# rgb[3:0] -> B[3:0]
set_property PACKAGE_PIN G14 [get_ports {rgb[3]}]
set_property PACKAGE_PIN H13 [get_ports {rgb[2]}]
set_property PACKAGE_PIN H12 [get_ports {rgb[1]}]
set_property PACKAGE_PIN G12 [get_ports {rgb[0]}]
set_property IOSTANDARD LVCMOS33 [get_ports {rgb[3] rgb[2] rgb[1] rgb[0]}]
